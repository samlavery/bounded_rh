theorem ee_mono {C s t : ℕ} {rest : Bool} (h : ee C s rest = true) (k : rest = true → C ≤ t)
    (hst : s ≤ t) : C ≤ t := by
  rcases ee_or h with h | h
  · exact h.trans hst
  · exact k h

theorem sdiv_le {a b : ℕ} {x y : ℝ} (ha : (a : ℝ) / SC ≤ x) (hb : (b : ℝ) / SC ≤ y) :
    ((a + b : ℕ) : ℝ) / SC ≤ x + y := by
  rw [Nat.cast_add, add_div]; exact add_le_add ha hb

theorem le_sdiv {a b : ℕ} {x y : ℝ} (ha : x ≤ (a : ℝ) / SC) (hb : y ≤ (b : ℝ) / SC) :
    x + y ≤ ((a + b : ℕ) : ℝ) / SC := by
  rw [Nat.cast_add, add_div]; exact add_le_add ha hb

theorem cut_bound {b : ℝ} {S : ℕ} {g : ℝ} (hS : (cN : ℝ) * SC ≤ b * S) (hcut : (S : ℝ) / SC ≤ g)
    (hb : 0 ≤ b) : (cN : ℝ) / SA ≤ (b / (SA:ℝ)) * g := by
  have hSA : (0:ℝ) < SA := by norm_num [SA]
  have hSC : (0:ℝ) < SC := by norm_num [SC]
  rw [div_mul_eq_mul_div, div_le_div_iff_of_pos_right hSA]
  have h1 : (cN : ℝ) ≤ b * ((S : ℝ) / SC) := by
    rw [mul_div_assoc', le_div_iff₀ hSC]; exact hS
  exact h1.trans (mul_le_mul_of_nonneg_left hcut hb)

theorem win_bound {a x : ℝ} (ha : (cN : ℝ) * 4 ≤ a) (hw : (1 / 4 : ℝ) ≤ wfun x) :
    (cN : ℝ) / SA ≤ (a / (SA:ℝ)) * wfun x := by
  have hSA : (0:ℝ) < SA := by norm_num [SA]
  rw [div_mul_eq_mul_div, div_le_div_iff_of_pos_right hSA]
  have ha0 : 0 ≤ a := le_trans (by positivity) ha
  linarith [mul_le_mul_of_nonneg_left hw ha0]
